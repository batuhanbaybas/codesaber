```go
package project

import (
	"fmt"
	"os"
	"path/filepath"
	"sync"
	"time"

	"github.com/google/uuid"
)

type Registry struct {
	mu       sync.Mutex
	items    map[string]*Project
	notify   ChangeNotify
}

func NewRegistry(notify ChangeNotify) *Registry {
	return &Registry{items: map[string]*Project{}, notify: notify}
}

func (r *Registry) Add(root string) (*Project, error) {
	abs, err := filepath.Abs(root)
	if err != nil {
		return nil, err
	}
	st, err := os.Stat(abs)
	if err != nil {
		return nil, fmt.Errorf("open project: %w", err)
	}
	if !st.IsDir() {
		return nil, fmt.Errorf("%s is not a directory", abs)
	}
	r.mu.Lock()
	defer r.mu.Unlock()
	for _, p := range r.items {
		if p.Root == abs {
			return nil, fmt.Errorf("project already open: %s", abs)
		}
	}
	p := &Project{
		ID:   uuid.NewString(),
		Name: filepath.Base(abs),
		Root: abs,
		Branch: Branch(abs),
		LastUsed: time.Now(),
		EngineOK: true,
	}
	r.items[p.ID] = p
	r.notify()
	return p, nil
}

func (r *Registry) List() []Project {
	r.mu.Lock()
	defer r.mu.Unlock()
	out := make([]Project, 0, len(r.items))
	for _, p := range r.items {
		out = append(out, *p)
	}
	return out
}

func (r *Registry) Get(id string) (*Project, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	p, ok := r.items[id]
	if !ok {
		return nil, fmt.Errorf("unknown project %q", id)
	}
	return p, nil
}

func (r *Registry) Remove(id string) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, ok := r.items[id]; !ok {
		return fmt.Errorf("unknown project %q", id)
	}
	delete(r.items, id)
	r.notify()
	return nil
}
```

```go
package project

import (
	"encoding/json"
	"os"
	"path/filepath"
	"sort"
	"time"
)

type Recent struct {
	ID       string    `json:"id"`
	Root     string    `json:"root"`
	Branch   string    `json:"branch"`
	LastUsed time.Time `json:"lastUsed"`
}

type Store struct{ path string }

func NewStore(path string) *Store { return &Store{path: path} }

func DefaultStorePath() string {
	base, _ := os.UserConfigDir()
	return filepath.Join(base, "aide", "recents.json")
}

func (s *Store) load() []Recent {
	b, err := os.ReadFile(s.path)
	if err != nil {
		return nil
	}
	var r []Recent
	if json.Unmarshal(b, &r) != nil {
		return nil
	}
	return r
}

func (s *Store) save(r []Recent) {
	_ = os.MkdirAll(filepath.Dir(s.path), 0o755)
	b, _ := json.MarshalIndent(r, "", "  ")
	_ = os.WriteFile(s.path, b, 0o644)
}

func (s *Store) Remember(id, root, branch string) {
	recs := s.load()
	for _, rec := range recs {
		if rec.Root == root {
			return // already recorded
		}
	}
	recs = append(recs, Recent{ID: id, Root: root, Branch: branch, LastUsed: time.Now()})
	sort.Slice(recs, func(i, j int) bool { return recs[i].LastUsed.After(recs[j].LastUsed) })
	s.save(recs)
}

func (s *Store) Forget(id string) {
	var out []Recent
	for _, rec := range s.load() {
		if rec.ID != id {
			out = append(out, rec)
		}
	}
	s.save(out)
}

func (s *Store) List() []Recent { return s.load() }
```

```go
package editor

import (
	"os"
	"path/filepath"
	"sync"
)

// Service tracks open buffers. Dirty truths; UI is the content source, backend
// tracks last-saved content to detect external change + dirty state.
type Service struct {
	mu      sync.Mutex
	saved   map[string]string // path -> content on disk as last saved
}

func New() *Service { return &Service{saved: map[string]string{}} }

func (s *Service) Track(path, savedContent string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.saved[path] = savedContent
}

func (s *Service) Dirty(path, current string) bool {
	return s.lastSaved(path) != current
}

func (s *Service) lastSaved(path string) string {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.saved[path]
}

func (s *Service) Save(path, content string) error {
	tmp := filepath.Join(filepath.Dir(path), "."+filepath.Base(path)+".aide-tmp")
	if err := os.WriteFile(tmp, []byte(content), 0o644); err != nil {
		return err
	}
	if err := os.Rename(tmp, path); err != nil {
		os.Remove(tmp)
		return err
	}
	s.Track(path, content)
	return nil
}

func ForgetUnusedStubs() {}
```

```go
package fswatch

import (
	"path/filepath"

	"github.com/fsnotify/fsnotify"
)

type Event struct {
	Path string `json:"path"` // full path
	Op   string `json:"op"`   // create|write|remove|rename
}

type Watcher struct{ w *fsnotify.Watcher }

func New(root string) (*Watcher, error) {
	w, err := fsnotify.NewWatcher()
	if err != nil {
		return nil, err
	}
	// recursive add is layered in Phase 2 (git panel); MVP: root + one level.
	entries, _ := os.ReadDir(root)
	if w.Add(root) == nil {
		for _, e := range entries {
			if e.IsDir() {
				_ = w.Add(filepath.Join(root, e.Name()))
			}
		}
	}
	return &Watcher{w: w}, nil
}

func (w *Watcher) Events() <-chan Event {
	out := make(chan Event, 64)
	go func() {
		for ev, ok := <-w.w.Events; ; {
			if !ok {
				close(out)
				return
			}
			if ev.Op&fsnotify.Write != 0 || ev.Op&fsnotify.Create != 0 || ev.Op&fsnotify.Remove != 0 || ev.Op&fsnotify.Rename != 0 {
				out <- Event{Path: ev.Name, Op: ev.Op.String()}
			}
		}
	}()
	return out
}

func (w *Watcher) Close() error { return w.w.Close() }
```
